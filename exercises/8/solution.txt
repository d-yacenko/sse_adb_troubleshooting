# Подключаемся под gpadmin на mdw.
sudo su - gpadmin

# В отдельной сессии должен быть запущен тестовый INSERT из задания.
# Он остается в состоянии выполнения и не завершается.

# Сначала проверяем общее состояние кластера.
gpstate
gpstate -e

# В этом кейсе обе команды могут показывать, что все сегменты работают нормально.
# Поэтому смотрим расширенную информацию по segment instances и репликации.
gpstate -s

# На проблемных mirror обратить внимание на:
#   WAL Sent Location
#   WAL Flush Location
#   WAL Replay Location
#   Bytes received but remain to flush
#   Bytes received but remain to replay
#
# Если WAL Sent Location ушел вперед, а Flush/Replay не продвигаются,
# повторяем проверку через несколько секунд.
sleep 10
gpstate -s

# В текущей конфигурации стенда два проблемных mirror находятся на sdw2.
# Подключаемся к этому segment-host.
SDW2=$(hostname -f | sed 's/-mdw/-sdw2/')
ssh "$SDW2"

# Проверяем процессы приема репликационного потока.
ps -eo pid,ppid,stat,etime,cmd | grep '[w]al receiver'

# В нормальном состоянии процессы wal receiver не должны иметь T в STAT.
# В проблемном состоянии оба процесса находятся в состоянии T (stopped).

# Получаем PID обоих остановленных процессов.
PIDS=$(ps -eo pid=,stat=,cmd= | awk '$2 ~ /T/ && /[w]al receiver process/ {print $1}' | xargs)

# Убеждаемся, что найдено два процесса.
echo "$PIDS"

# Возобновляем работу обоих wal receiver.
kill -CONT $PIDS

# Проверяем, что состояние T исчезло.
ps -eo pid,ppid,stat,etime,cmd | grep '[w]al receiver'

# Возвращаемся на mdw.
exit

# Проверяем, что mirror снова продвигают WAL.
gpstate -s

# После восстановления WAL Flush/Replay должны догнать WAL Sent,
# а Bytes received but remain to flush/replay вернуться к нулю.
#
# Если исходный INSERT не был отменен пользователем, он должен завершиться
# после восстановления обоих wal receiver.

# Контрольная проверка данных.
psql adb -c "SELECT count(*) FROM public.write_test;"

# Дополнительно выполняем новую операцию записи.
psql adb -c "
INSERT INTO public.write_test
SELECT
    i + 100000,
    repeat(md5(i::text), 16)
FROM generate_series(1,10000) i;
"

# Финальная проверка состояния.
gpstate
gpstate -s

# Альтернативный вариант восстановления:
# вместо SIGCONT можно завершить оба остановленных wal receiver:
#
# kill -9 <PID1> <PID2>
#
# postmaster создаст новые процессы wal receiver.
# Полный перезапуск кластера для устранения этой неисправности не требуется.
