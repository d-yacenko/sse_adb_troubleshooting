# проверяем общую работоспособность кластера
sudo su - gpadmin

gpstate -a
psql adb -c "select now();"


# проверяем используемый тип interconnect
psql adb -c "show gp_interconnect_type;"
gpconfig -s gp_interconnect_type


# при необходимости проверяем сетевые настройки хостов кластера
gpssh -f arenadata_configs/arenadata_all_hosts.hosts -e "ip -o link show | grep -v 'lo:'"


# меняем тип interconnect
gpconfig -c gp_interconnect_type -v tcp

# параметр требует перезапуска кластера
gpstop -ra


# проверяем состояние кластера после перезапуска
gpstate -a
psql adb -c "show gp_interconnect_type;"
gpconfig -s gp_interconnect_type


# повторно выполняем исходный запрос
psql adb

EXPLAIN ANALYZE
SELECT
    a.join_key,
    count(*),
    sum(length(a.payload) + length(b.payload))
FROM motion_a a
JOIN motion_b b
    ON a.join_key = b.join_key
GROUP BY a.join_key;
